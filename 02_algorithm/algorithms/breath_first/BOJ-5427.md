# 문제 정리

- **문제명 / 번호**: BOJ - 5427 // 불
- **유형**: BFS
- **핵심 키워드**: 불과 사람의 위치를 구분하는 BFS를 구현
- **제한 보고 떠올릴 것**: 큐에 append 하는 순서를 주의해야한다.
- **상태 정의**: 해당 좌표에 불이 있는지 없는지, 좌표에 있는게 사람인지, 사람이라면 그 곳까지 가는데 시간이 얼마나 걸렸는지
- **핵심 아이디어**: 큐 내부 상태를 4개를 넣자. (r, c, s: 불 or 사람, t: if 사람, 걸린 시간)
- **시간복잡도**: 
- **실수 포인트**: 처음에 dist 배열을 만들어서 시간을 따로 관리하려 했는데 현재 사람의 위치에 불이 올 수 있어서 따로 bfs 파라미터로 관리하는 방향으로 수정했다.

```python
            return False

            return False

            return False
```

- **AI 정석 코드**
```python
from collections import deque
import sys
input = sys.stdin.readline

DIRECTIONS = [(-1, 0), (1, 0), (0, -1), (0, 1)]


def spread_fire(building, fires, H, W):
    fire_time = [[-1] * W for _ in range(H)]
    q = deque()

    for r, c in fires:
        q.append((r, c))
        fire_time[r][c] = 0

    while q:
        r, c = q.popleft()

        for dr, dc in DIRECTIONS:
            nr, nc = r + dr, c + dc

            if not (0 <= nr < H and 0 <= nc < W):
                continue
            if building[nr][nc] == '#':
                continue
            if fire_time[nr][nc] != -1:
                continue

            fire_time[nr][nc] = fire_time[r][c] + 1
            q.append((nr, nc))

    return fire_time


def escape(building, start, fire_time, H, W):
    visited = [[-1] * W for _ in range(H)]
    q = deque()

    sr, sc = start
    q.append((sr, sc))
    visited[sr][sc] = 0

    while q:
        r, c = q.popleft()
        cur_time = visited[r][c]

        for dr, dc in DIRECTIONS:
            nr, nc = r + dr, c + dc
            next_time = cur_time + 1

            # 맵 밖으로 나가면 탈출 성공
            if not (0 <= nr < H and 0 <= nc < W):
                return next_time

            if building[nr][nc] == '#':
                continue
            if visited[nr][nc] != -1:
                continue

            # next_time에 도착하려는데, 불이 그보다 먼저 또는 같은 시간에 오면 못 감
            if fire_time[nr][nc] != -1 and fire_time[nr][nc] <= next_time:
                continue

            visited[nr][nc] = next_time
            q.append((nr, nc))

    return "IMPOSSIBLE"


TC = int(input())
for _ in range(TC):
    W, H = map(int, input().split())

    building = []
    fires = []
    start = None

    for r in range(H):
        row = input().strip()
        building.append(row)

        for c, ch in enumerate(row):
            if ch == '*':
                fires.append((r, c))
            elif ch == '@':
                start = (r, c)

    fire_time = spread_fire(building, fires, H, W)
    print(escape(building, start, fire_time, H, W))
```
